# 07 · 技术架构与工程规范

> 本章让 AI 开发者搭建工程骨架并本地跑通。

## 1. 技术架构图

```
浏览器（web-user 用户端 + web-admin 管理端，React 18 + AntD 5 + Vite）
   │ HTTP/JSON（/api 代理）
   ▼
后端 API (Node.js 18 + Express 4)
   ├── 路由/控制器
   ├── 业务服务
   │    ├── Agent 编排（Orchestrator + 6 Agents）
   │    ├── LLM Provider（deepseek / mock）
   │    ├── PDF 提取（pdf-parse）
   │    ├── 存储适配器（local FS / MinIO 可选）
   │    └── 报告生成（pdfkit）
   ├── MySQL 8.0（mysql2，库 bid_demo）
   │      ├── 云端：云 MySQL(RDS) / Docker MySQL + 定时 mysqldump 上云
   │      └── 本地：本地 MySQL（自动装配兜底）
   └── 本地文件存储（server/storage/uploads）               ← Docker 卷 uploads-data（可上云）
```

## 6. 数据库迁移脚本（scripts/migrate.js）

```js
const mysql = require('mysql2/promise');
const fs = require('fs');
async function migrate() {
  const conn = await mysql.createConnection({
    host: process.env.DB_HOST,
    port: Number(process.env.DB_PORT || 3306),
    user: process.env.DB_USER,
    password: process.env.DB_PASSWORD,
    multipleStatements: true,
  });
  await conn.query(fs.readFileSync('./schema.sql', 'utf8'));
  await conn.end();
  console.log('DB migrated');
}
migrate();
```

## 8. 存储适配器（storageAdapter.js）

```js
// 统一接口：save(file, destName) -> {path, url}; remove(path)
class LocalStorage {
  save(fileBuffer, destName) {
    const p = path.join(STORAGE_DIR, destName);
    fs.writeFileSync(p, fileBuffer);
    return { path: destName, url: `/uploads/${destName}` };
  }
}
// MinIOStorage：生产可选，MVP 不实现
```

- `app.use('/uploads', express.static(STORAGE_DIR))` 提供文件访问。

## 9. 鉴权中间件（middleware/auth.js）

```js
function auth(req, res, next) {
  const h = req.headers.authorization || '';
  const token = h.startsWith('Bearer ') ? h.slice(7) : null;
  try { req.user = jwt.verify(token, JWT_SECRET); next(); }
  catch { res.status(401).json({ code: 401, message: '未登录或 token 无效', data: null }); }
}
function requireRole(...roles) {
  return (req, res, next) => roles.includes(req.user.role)
    ? next() : res.status(403).json({ code: 403, message: '无权限', data: null });
}
```

## 11. 启动流程

**方式一：本地直接开发（自动装配数据库）**

```bash
# 1. 后端
cd server
cp .env.example .env        # 如需真实 LLM 填 KEY，否则默认 mock
npm install
npm run db:provision        # 自动装配 MySQL：云 → Docker → 本地 → 自动安装
npm run db:migrate          # 执行 schema.sql 建表
npm run seed                # 种子用户
npm run dev                 # http://localhost:3001

# 2. 前端（两个应用，各开终端）
cd web-user && npm install && npm run dev    # 用户端 http://localhost:5173（/api 代理到 3001）
cd web-admin && npm install && npm run dev   # 管理端 http://localhost:5174（/api 代理到 3001）
```

**方式二：Docker 一键启动（MySQL + 后端 + 前端，数据可上云）**

```bash
docker compose up -d --build
# 用户端 http://localhost:8080 / 管理端 http://localhost:8081（nginx 托管，/api、/uploads 反代到后端）
# 后端 http://localhost:3001；MySQL 数据挂载卷 mysql-data，上传文件挂载卷 uploads-data；云端同步见「数据存储方式」
```

**web-user/nginx.conf（web-admin 同结构）**

```nginx
server {
  listen 80;
  root /usr/share/nginx/html;
  index index.html;

  # SPA 路由回退
  location / {
    try_files $uri $uri/ /index.html;
  }

  # API 反向代理到后端
  location /api/ {
    proxy_pass http://server:3001;
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
  }

  # 上传文件反向代理到后端
  location /uploads/ {
    proxy_pass http://server:3001;
  }
}
```

**docker-compose.yml（仓库根）**

```yaml
services:
  mysql:
    image: mysql:8.0
    restart: unless-stopped
    environment:
      MYSQL_ROOT_PASSWORD: root123456
      MYSQL_DATABASE: bid_demo
    ports:
      - "3306:3306"
    volumes:
      - mysql-data:/var/lib/mysql          # MySQL 数据持久化
    healthcheck:
      test: ["CMD", "mysqladmin", "ping", "-h", "localhost"]
      interval: 5s
      timeout: 5s
      retries: 10

  server:
    build: ./server
    depends_on:
      mysql:
        condition: service_healthy
    ports:
      - "3001:3001"
    environment:
      PORT: 3001
      DB_HOST: mysql
      DB_PORT: 3306
      DB_USER: root
      DB_PASSWORD: root123456
      DB_NAME: bid_demo
      STORAGE_DIR: /app/storage/uploads
      STORAGE_DRIVER: local
      LLM_MOCK: "true"
      JWT_SECRET: docker-dev-secret
    volumes:
      - uploads-data:/app/storage           # 上传文件持久化
    command: sh -c "node scripts/migrate.js && node seed.js && node src/index.js"

  web-user:
    build: ./web-user
    depends_on:
      - server
    ports:
      - "8080:80"

  web-admin:
    build: ./web-admin
    depends_on:
      - server
    ports:
      - "8081:80"

volumes:
  mysql-data:
  uploads-data:
```

- **数据持久化说明**：MySQL 数据写入卷 `mysql-data`（容器内 `/var/lib/mysql`），上传文件写入卷 `uploads-data`（容器内 `/app/storage/uploads`）；容器重建数据不丢失。
- **前端托管**：`web-user` / `web-admin` 两个服务均用多阶段构建（`npm run build` → nginx 托管 `dist`），`nginx.conf` 将 `/api`、`/uploads` 反向代理到 `server:3001`；本地开发仍可用 Vite dev（代理到 3001）。

## 14. 数据存储方式（三级策略）

> 目标：优先把系统运行数据存到云端；云端不可行时用本地 MySQL；本地无 MySQL 时自动安装。

### 14.1 三级策略

| 优先级 | 场景 | 实现 |
|:---:|------|------|
| 1 | 云端存储（目标态） | ① 配置 `CLOUD_DB_URL` 直连云 MySQL（RDS）；② 或 Docker 部署 MySQL，由容器内定时任务 `mysqldump` 上传到云端对象存储（S3 兼容/OSS/MinIO 云端） |
| 2 | 本地 MySQL（兜底） | Docker 云端不可达或未配置云端时，使用本地 MySQL 8.0（`127.0.0.1:3306`） |
| 3 | 自动装配 | 本地无 MySQL 时，`db:provision` 自动安装并初始化 |

### 14.2 数据库自动装配（scripts/provision-db.js）

```
1. 若 CLOUD_DB_URL 已配置 → 解析直连云 MySQL，跳过本地准备
2. 否则若检测到 Docker 可用 → docker compose --project-directory .. up -d mysql（含健康检查）
3. 否则检测本地 MySQL（3306 端口 / mysql 命令）→ 使用
4. 否则自动安装 MySQL：
   - Windows：winget install Oracle.MySQL 或下载 MSI 静默安装
   - macOS：brew install mysql
   - Linux：apt-get/yum install mysql-server
   - 安装后启动服务，创建 DB 与用户（root/root123456）
5. 运行 db:migrate 建表
```

### 14.3 云端同步（scripts/cloud-sync.js / cloud-restore.js）

- `cloud:sync`：容器内定时执行 `mysqldump bid_demo` → 压缩 → 上传到云端对象存储。
- `cloud:restore`：从云端下载最新 dump → `mysql < dump.sql` 恢复。
- 未配置 `CLOUD_STORAGE_DRIVER` 时，云端同步禁用，数据仅保存在本地/MySQL 卷。

**云端对象存储 SDK 选型（按 `CLOUD_STORAGE_DRIVER` 分派）**：

| driver | SDK | 说明 |
|--------|-----|------|
| `s3` | `@aws-sdk/client-s3` | AWS S3 及任意 S3 兼容协议（含 MinIO、部分 OSS 的 S3 兼容模式） |
| `oss` | `ali-oss` | 阿里云 OSS 原生 SDK |
| `minio` | `minio` | MinIO 专用 SDK |

**存储驱动抽象（services/storage/cloudStorage.js）**：

```js
// 统一接口：upload(key, data) / download(key) -> data
function createCloudStorage() {
  switch (process.env.CLOUD_STORAGE_DRIVER) {
    case 's3':    return new S3Storage();      // @aws-sdk/client-s3
    case 'oss':   return new OssStorage();     // ali-oss
    case 'minio': return new MinioStorage();   // minio
    default:      return null;                 // 未配置 → 云端同步禁用
  }
}
```
